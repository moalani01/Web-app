<template>
  <div class="spinner-container" :class="sizeClass">
    <div class="spinner" :style="spinnerStyle">
      <svg viewBox="0 0 24 24" fill="none" xmlns="http://www.w3.org/2000/svg">
        <circle
          cx="12"
          cy="12"
          r="10"
          stroke="currentColor"
          stroke-width="2"
          stroke-linecap="round"
          stroke-dasharray="60"
          stroke-dashoffset="15"
        />
      </svg>
    </div>
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue'

interface Props {
  size?: 'sm' | 'md' | 'lg'
  color?: string
}

const props = withDefaults(defineProps<Props>(), {
  size: 'md',
  color: 'hsl(var(--primary))'
})

const sizeClass = computed(() => `spinner-${props.size}`)

const spinnerStyle = computed(() => ({
  color: props.color
}))
</script>

<style scoped>
.spinner-container {
  display: inline-flex;
  align-items: center;
  justify-content: center;
}

.spinner {
  animation: spin 1s linear infinite;
}

.spinner-sm .spinner {
  width: 1rem;
  height: 1rem;
}

.spinner-md .spinner {
  width: 1.5rem;
  height: 1.5rem;
}

.spinner-lg .spinner {
  width: 2rem;
  height: 2rem;
}

@keyframes spin {
  from {
    transform: rotate(0deg);
  }
  to {
    transform: rotate(360deg);
  }
}
</style>
