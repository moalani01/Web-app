<template>
  <div class="radio-group" role="radiogroup">
    <div v-for="option in options" :key="option" class="radio-item-container">
      <button
        :id="`radio-${option}`"
        role="radio"
        :aria-checked="modelValue === option"
        class="radio-item"
        :class="{ checked: modelValue === option }"
        @click="selectOption(option)"
      >
        <span v-if="modelValue === option" class="radio-indicator" />
      </button>
      <label :for="`radio-${option}`" class="radio-label" @click="selectOption(option)">
        {{ option }}
      </label>
    </div>
  </div>
</template>

<script setup lang="ts">
interface Props {
  modelValue?: string
  options: string[]
}

defineProps<Props>()

const emit = defineEmits<{
  'update:modelValue': [value: string]
}>()

const selectOption = (option: string) => {
  emit('update:modelValue', option)
}
</script>
