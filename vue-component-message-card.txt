<template>
  <div class="message-type-item">
    <BaseButton
      :variant="isSelected ? 'default' : 'outline'"
      class="message-type-button"
      @click="$emit('select')"
    >
      <div class="button-content">
        <div class="button-title">{{ messageType.name }}</div>
        <div class="button-description" :class="{ expanded: isExpanded }">
          {{ displayText }}
        </div>
      </div>
    </BaseButton>
    
    <BaseButton
      v-if="hasLongDescription"
      variant="ghost"
      size="sm"
      class="expand-button"
      @click="isExpanded = !isExpanded"
    >
      {{ isExpanded ? 'Show less' : 'Show more' }}
    </BaseButton>
  </div>
</template>

<script setup lang="ts">
import { ref, computed } from 'vue'
import type { MessageType } from '@/types/iot'
import BaseButton from '@/components/common/BaseButton.vue'
import { truncateText, shouldTruncate } from '@/utils/helpers'
import { CONFIG } from '@/config/constants'

const props = defineProps<{
  messageType: MessageType
  isSelected: boolean
}>()

defineEmits<{
  select: []
}>()

const isExpanded = ref(false)

const hasLongDescription = computed(() => 
  shouldTruncate(props.messageType.description, CONFIG.TEXT.MAX_DESCRIPTION_LENGTH)
)

const displayText = computed(() => {
  if (isExpanded.value || !hasLongDescription.value) {
    return props.messageType.description
  }
  return truncateText(props.messageType.description, CONFIG.TEXT.MAX_DESCRIPTION_LENGTH)
})
</script>
