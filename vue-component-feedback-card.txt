<template>
  <div class="feedback-item">
    <div class="feedback-content">
      <div class="feedback-icon">
        <svg v-if="feedback.type === 'success'" class="icon icon-success" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
          <path d="M22 11.08V12a10 10 0 1 1-5.93-9.14"></path>
          <polyline points="22 4 12 14.01 9 11.01"></polyline>
        </svg>
        <svg v-else-if="feedback.type === 'error'" class="icon icon-error" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
          <circle cx="12" cy="12" r="10"></circle>
          <line x1="15" y1="9" x2="9" y2="15"></line>
          <line x1="9" y1="9" x2="15" y2="15"></line>
        </svg>
        <svg v-else class="icon icon-info" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
          <circle cx="12" cy="12" r="10"></circle>
          <polyline points="12 6 12 12 16 14"></polyline>
        </svg>
      </div>
      
      <div class="feedback-text">
        <div class="feedback-message">{{ displayMessage }}</div>
        <div v-if="displayDetails" class="feedback-details">{{ displayDetails }}</div>
        
        <BaseButton
          v-if="hasLongContent"
          variant="ghost"
          size="sm"
          class="expand-button"
          @click="isExpanded = !isExpanded"
        >
          {{ isExpanded ? 'Show less' : 'Show more' }}
        </BaseButton>
        
        <div class="feedback-timestamp">{{ formattedTimestamp }}</div>
      </div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, computed } from 'vue'
import type { FeedbackMessage } from '@/types/iot'
import BaseButton from '@/components/common/BaseButton.vue'
import { truncateText, shouldTruncate } from '@/utils/helpers'
import { CONFIG } from '@/config/constants'

const props = defineProps<{
  feedback: FeedbackMessage
}>()

const isExpanded = ref(false)

const hasLongContent = computed(() => 
  shouldTruncate(props.feedback.message, CONFIG.TEXT.MAX_TRUNCATE_LENGTH) ||
  (props.feedback.details ? shouldTruncate(props.feedback.details, CONFIG.TEXT.MAX_TRUNCATE_LENGTH) : false)
)

const displayMessage = computed(() => {
  if (isExpanded.value || !shouldTruncate(props.feedback.message, CONFIG.TEXT.MAX_TRUNCATE_LENGTH)) {
    return props.feedback.message
  }
  return truncateText(props.feedback.message, CONFIG.TEXT.MAX_TRUNCATE_LENGTH)
})

const displayDetails = computed(() => {
  if (!props.feedback.details) return null
  if (isExpanded.value || !shouldTruncate(props.feedback.details, CONFIG.TEXT.MAX_TRUNCATE_LENGTH)) {
    return props.feedback.details
  }
  return truncateText(props.feedback.details, CONFIG.TEXT.MAX_TRUNCATE_LENGTH)
})

const formattedTimestamp = computed(() => 
  props.feedback.timestamp.toLocaleTimeString()
)
</script>
