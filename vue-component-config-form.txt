<template>
  <BaseCard class="configuration-card">
    <template #header>
      <h2 class="card-title">
        {{ currentMessageType ? currentMessageType.name : 'Select Message Type' }}
      </h2>
    </template>
    <template #content>
      <div v-if="currentMessageType" class="form-container">
        <BaseScrollArea class="form-content">
          <div class="form-fields">
            <div v-for="field in currentMessageType.fields" :key="field.name" class="field-container">
              <div class="field-header">
                <label class="field-label" :for="`field-${field.name}`">
                  {{ field.label }}
                  <span v-if="field.required" class="required-mark">*</span>
                </label>
                <BaseSwitch
                  v-if="field.type === 'boolean'"
                  :id="`field-${field.name}`"
                  :value="formData[field.name] ?? field.defaultValue"
                  @update:value="$emit('field-change', field.name, $event)"
                />
              </div>
              <component
                v-if="field.type !== 'boolean'"
                :is="getFieldComponent(field)"
                :id="`field-${field.name}`"
                :value="formData[field.name] ?? field.defaultValue"
                @update:value="$emit('field-change', field.name, $event)"
                v-bind="getFieldProps(field)"
              />
            </div>
          </div>
        </BaseScrollArea>
        
        <div class="form-footer">
          <BaseButton 
            @click="$emit('send-message')"
            :disabled="!isConnected"
            class="send-button"
          >
            <svg class="icon-send" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round">
              <line x1="22" y1="2" x2="11" y2="13"></line>
              <polygon points="22 2 15 22 11 13 2 9 22 2"></polygon>
            </svg>
            Send Configuration
          </BaseButton>
        </div>
      </div>
      <div v-else class="empty-state">
        Please select a message type to configure
      </div>
    </template>
  </BaseCard>
</template>

<script setup lang="ts">
import type { MessageType, MessageField } from '@/types/iot'
import BaseCard from '@/components/common/BaseCard.vue'
import BaseButton from '@/components/common/BaseButton.vue'
import BaseScrollArea from '@/components/common/BaseScrollArea.vue'
import BaseInput from '@/components/common/BaseInput.vue'
import BaseSelect from '@/components/common/BaseSelect.vue'
import BaseSwitch from '@/components/common/BaseSwitch.vue'
import BaseRadioGroup from '@/components/common/BaseRadioGroup.vue'

defineProps<{
  currentMessageType?: MessageType
  formData: Record<string, any>
  isConnected: boolean
}>()

defineEmits<{
  'field-change': [fieldName: string, value: any]
  'send-message': []
}>()

const getFieldComponent = (field: MessageField) => {
  const componentMap = {
    text: BaseInput,
    number: BaseInput,
    dropdown: BaseSelect,
    boolean: BaseSwitch,
    radio: BaseRadioGroup
  }
  return componentMap[field.type] || BaseInput
}

const getFieldProps = (field: MessageField) => {
  const baseProps: any = {
    placeholder: `Enter ${field.label.toLowerCase()}`
  }

  if (field.type === 'number') {
    return { ...baseProps, type: 'number' }
  }
  
  if (field.type === 'dropdown' || field.type === 'radio') {
    return { 
      options: field.options || [],
      placeholder: field.type === 'dropdown' ? `Select ${field.label.toLowerCase()}` : undefined
    }
  }
  
  return baseProps
}
</script>
