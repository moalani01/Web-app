<template>
  <div class="feedback-item">
    <div class="feedback-content">
      <div class="feedback-icon">
        <component :is="icon" :class="iconClass" />
      </div>
      
      <div class="feedback-text">
        <div class="feedback-message">
          {{ displayMessage }}
        </div>
        
        <div v-if="displayDetails" class="feedback-details">
          {{ displayDetails }}
        </div>
        
        <BaseButton
          v-if="shouldShowExpandButton"
          variant="ghost"
          size="sm"
          class="expand-button"
          @click="$emit('toggle-expand')"
        >
          {{ isExpanded ? 'Show less' : 'Show more' }}
        </BaseButton>
        
        <div class="feedback-timestamp">
          {{ formattedTimestamp }}
        </div>
      </div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue'
import type { FeedbackMessage, FeedbackType } from '@/types/iot'
import BaseButton from '@/components/common/BaseButton.vue'
import { useTextUtils } from '@/composables/useTextUtils'

interface Props {
  feedback: FeedbackMessage
  isExpanded: boolean
}

interface Emits {
  (e: 'toggle-expand'): void
}

const props = defineProps<Props>()
defineEmits<Emits>()

const { shouldTruncate, getDisplayText } = useTextUtils()

const FEEDBACK_STYLES: Record<FeedbackType, string> = {
  success: 'icon-success',
  error: 'icon-error',
  info: 'icon-info'
}

const icon = computed(() => {
  const icons = {
    success: CheckCircleIcon,
    error: XCircleIcon,
    info: ClockIcon
  }
  return icons[props.feedback.type]
})

const iconClass = computed(() => FEEDBACK_STYLES[props.feedback.type])

const shouldTruncateMessage = computed(() => 
  shouldTruncate(props.feedback.message, 100)
)

const shouldTruncateDetails = computed(() => 
  props.feedback.details && shouldTruncate(props.feedback.details, 100)
)

const shouldShowExpandButton = computed(() => 
  shouldTruncateMessage.value || shouldTruncateDetails.value
)

const displayMessage = computed(() => 
  getDisplayText(props.feedback.message, props.isExpanded, 100)
)

const displayDetails = computed(() => 
  props.feedback.details 
    ? getDisplayText(props.feedback.details, props.isExpanded, 100)
    : undefined
)

const formattedTimestamp = computed(() => 
  props.feedback.timestamp.toLocaleTimeString()
)

// Icon components
const CheckCircleIcon = {
  template: `
    <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" class="icon">
      <path d="M22 11.08V12a10 10 0 1 1-5.93-9.14"></path>
      <polyline points="22 4 12 14.01 9 11.01"></polyline>
    </svg>
  `
}

const XCircleIcon = {
  template: `
    <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" class="icon">
      <circle cx="12" cy="12" r="10"></circle>
      <line x1="15" y1="9" x2="9" y2="15"></line>
      <line x1="9" y1="9" x2="15" y2="15"></line>
    </svg>
  `
}

const ClockIcon = {
  template: `
    <svg viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round" class="icon">
      <circle cx="12" cy="12" r="10"></circle>
      <polyline points="12 6 12 12 16 14"></polyline>
    </svg>
  `
}
</script>
