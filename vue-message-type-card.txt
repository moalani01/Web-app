<template>
  <div class="message-type-item">
    <BaseButton
      :variant="isSelected ? 'default' : 'outline'"
      class="message-type-button"
      @click="$emit('select')"
    >
      <div class="button-content">
        <div class="button-title">{{ messageType.name }}</div>
        <div class="button-description" :class="{ 'description-expanded': isExpanded }">
          {{ displayText }}
        </div>
      </div>
    </BaseButton>
    
    <BaseButton
      v-if="hasLongDescription"
      variant="ghost"
      size="sm"
      class="expand-button"
      @click="$emit('toggle-expand')"
    >
      {{ isExpanded ? 'Show less' : 'Show more' }}
    </BaseButton>
  </div>
</template>

<script setup lang="ts">
import { computed } from 'vue'
import type { MessageType } from '@/types/iot'
import BaseButton from '@/components/common/BaseButton.vue'
import { useTextUtils } from '@/composables/useTextUtils'

interface Props {
  messageType: MessageType
  isSelected: boolean
  isExpanded: boolean
}

interface Emits {
  (e: 'select'): void
  (e: 'toggle-expand'): void
}

const props = defineProps<Props>()
defineEmits<Emits>()

const { shouldTruncate, getDisplayText } = useTextUtils()

const hasLongDescription = computed(() => 
  shouldTruncate(props.messageType.description)
)

const displayText = computed(() => 
  getDisplayText(props.messageType.description, props.isExpanded)
)
</script>
