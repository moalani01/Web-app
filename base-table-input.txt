<template>
  <div class="table-input-container">
    <div class="table-wrapper">
      <table class="table-input">
        <thead>
          <tr>
            <th v-for="column in columns" :key="column.name" :style="{ width: column.width }">
              {{ column.label }}
            </th>
          </tr>
        </thead>
        <tbody>
          <tr v-for="(row, rowIndex) in tableData" :key="rowIndex">
            <td v-for="column in columns" :key="`${rowIndex}-${column.name}`">
              <BaseInput
                v-if="column.type === 'text' || column.type === 'number'"
                :type="column.type"
                :modelValue="row[column.name]"
                @update:modelValue="updateCell(rowIndex, column.name, $event)"
                :placeholder="`Enter ${column.label.toLowerCase()}`"
                class="table-cell-input"
              />
              <BaseSelect
                v-else-if="column.type === 'dropdown'"
                :modelValue="row[column.name]"
                @update:modelValue="updateCell(rowIndex, column.name, $event)"
                :options="column.options || []"
                :placeholder="`Select ${column.label.toLowerCase()}`"
                class="table-cell-select"
              />
            </td>
          </tr>
        </tbody>
      </table>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, watch } from 'vue'
import type { TableColumn } from '@/types/iot'
import BaseInput from './BaseInput.vue'
import BaseSelect from './BaseSelect.vue'

interface Props {
  modelValue: Record<string, any>[]
  columns: TableColumn[]
  rows: number
}

const props = withDefaults(defineProps<Props>(), {
  rows: 7
})

const emit = defineEmits<{
  'update:modelValue': [value: Record<string, any>[]]
}>()

// Initialize table data
const initializeTableData = () => {
  const data = []
  for (let i = 0; i < props.rows; i++) {
    const row: Record<string, any> = {}
    props.columns.forEach(column => {
      row[column.name] = props.modelValue?.[i]?.[column.name] || ''
    })
    data.push(row)
  }
  return data
}

const tableData = ref(initializeTableData())

// Update cell value
const updateCell = (rowIndex: number, columnName: string, value: any) => {
  tableData.value[rowIndex][columnName] = value
  emit('update:modelValue', [...tableData.value])
}

// Watch for external changes to modelValue
watch(() => props.modelValue, (newValue) => {
  if (newValue && Array.isArray(newValue)) {
    tableData.value = initializeTableData()
  }
}, { deep: true })
</script>
