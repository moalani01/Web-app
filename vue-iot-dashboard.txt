<template>
  <div class="dashboard-container">
    <div class="dashboard-content">
      <DashboardHeader :is-connected="isConnected" />

      <div class="dashboard-grid">
        <MessageTypeList
          :message-types="MESSAGE_TYPES"
          :selected-message-type="selectedMessageType"
          @select-message-type="setSelectedMessageType"
        />

        <ConfigurationForm
          :current-message-type="currentMessageType"
          :form-data="formData"
          :is-connected="isConnected"
          @field-change="handleFieldChange"
          @send-message="handleSendMessage"
        />

        <FeedbackDisplay
          :feedback-messages="feedbackMessages"
        />
      </div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, computed, onUnmounted } from 'vue'
import { MESSAGE_TYPES } from '@/constants/messageTypes'
import { CONFIG } from '@/config/constants'
import type { FeedbackMessage } from '@/types/iot'
import { createFeedbackMessage } from '@/utils/feedbackHelpers'
import { useDebouncedFn } from '@/composables/useDebounce'

// Components
import DashboardHeader from './DashboardHeader.vue'
import MessageTypeList from './MessageTypeList.vue'
import ConfigurationForm from './ConfigurationForm.vue'
import FeedbackDisplay from './FeedbackDisplay.vue'

// Composables
import { useLocalStorage } from '@/composables/useLocalStorage'
import { useConnectionStatus } from '@/composables/useConnectionStatus'
import { useFeedback } from '@/composables/useFeedback'
import { useMessageSender } from '@/composables/useMessageSender'

// State
const selectedMessageType = ref<string>('')
const { value: formData, setValue: setFormData } = useLocalStorage<Record<string, any>>(
  CONFIG.STORAGE_KEYS.IOT_CONFIG_DATA, 
  {}
)

// Composables
const isConnected = useConnectionStatus()
const { messages: feedbackMessages, addMessage: addFeedbackMessage } = useFeedback()

// Computed
const currentMessageType = computed(() => 
  MESSAGE_TYPES.find(mt => mt.id === selectedMessageType.value)
)

// Debounced form update to improve performance
const debouncedSetFormData = useDebouncedFn(setFormData, 300)

// Methods
const setSelectedMessageType = (id: string) => {
  selectedMessageType.value = id
}

const handleFieldChange = (fieldName: string, value: any) => {
  const newFormData = {
    ...formData.value,
    [fieldName]: value
  }
  // Update local state immediately for UI responsiveness
  formData.value = newFormData
  // Debounce localStorage update
  debouncedSetFormData(newFormData)
}

const handleSuccessResponse = (response: any) => {
  const feedbackMessage = createFeedbackMessage(
    response.type,
    response.message,
    response.details
  )
  addFeedbackMessage(feedbackMessage)
}

const handleErrorResponse = (errorMessage: FeedbackMessage) => {
  addFeedbackMessage(errorMessage)
}

// Message sender
const { sendMessage } = useMessageSender(handleSuccessResponse, handleErrorResponse)

const handleSendMessage = async () => {
  if (!currentMessageType.value) return
  await sendMessage(currentMessageType.value, formData.value)
}

// Cleanup on unmount
onUnmounted(() => {
  // Any cleanup if needed
})
